% \pl{indent}{text}: one line of a poem; wrapped lines hang under it.
\newcommand{\pl}[2]{{\hangindent=\dimexpr#1+2em\relax\hangafter=1\hspace*{#1}#2\par}}
% \stanza{lines}: gap before a stanza, kept together where possible.
\newcommand{\stanza}[1]{\vspace{0.75\baselineskip}\needspace{#1\baselineskip}}
% \poemfooter{text}: bottom right of the poem's last page.
\newcommand{\poemfooter}[1]{\par\vfill\hfill{\color{muted}\small #1}\par}

% \cover{series}{subtitle}{author}{meta}{epigraph}
\newcommand{\cover}[5]{%
  \thispagestyle{empty}
  \null\vspace{0.33\textheight}
  {\fontsize{30}{36}\selectfont\color{accent}#1\par}
  \vspace{10pt}
  {\fontsize{17}{22}\selectfont #2\par}
  \vspace{12pt}
  {\fontsize{11}{14}\selectfont #3\par}
  \vspace{16pt}
  {\color{muted}\small #4\par}
  \vspace{36pt}
  {\color{muted}\itshape\fontsize{10.5}{16}\selectfont #5\par}
  \vfill
  \newpage}

\newcommand{\colophon}[1]{%

  {\centering\color{muted}\footnotesize #1\par}
  \vfill}
